namespace = frontera_ot_homeland
frontera_ot_homeland.1 = {
	type = state_event
	placement = ROOT

	title = frontera_ot_homeland.1.t
	desc = frontera_ot_homeland.1.d
	flavor = frontera_ot_homeland.1.f

	event_image = {
		video = "europenorthamerica_gold_prospectors"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	duration = 3

	cooldown = { days = -1 }

	trigger = {
		#triggered by event
	}

	immediate = {
		save_scope_as = ascendent_state
		owner = {
			save_scope_as = subject_scope
			owner.overlord = {
				save_scope_as = overlord_scope
			}
		}
	}

	cancellation_trigger = {
		scope:subject_scope = {
			OR = {
				is_diplomatic_play_committed_participant = yes
				NOT = { has_law = law_type:law_frontier_territorial_administration }
				NOT = { is_direct_subject_of = scope:overlord_scope }
			}
		}
	}

	option = {
		name = frontera_ot_homeland.1.a
		ai_chance = {
			base = 1 # subtract 1 
		}
		trigger = {
			scope:subject_scope = {
				OR = {
					is_ai = yes
					any_scope_state = {
						count >= 2
					}
				}
			}
		}
		default_option = yes
		custom_tooltip = {
			text = frontera_ot_homeland_gained_homeland
			scope:overlord_scope = {
				trigger_event = frontera_ot_homeland.2
			}
		}
	}
	option = {
		name = frontera_ot_homeland.1.c
		ai_chance = {
			base = 0 # AI will never do this
		}
		custom_tooltip = {
			text = frontera_ot_homeland_not_ready_for_statehood_tt
			scope:ascendent_state = {
				set_variable = {
					name = not_ready_for_statehood_frontera
					years = 3
				}
			}
		}
	}
	option = {
		name = frontera_ot_homeland.1.b
		trigger = {
			scope:subject_scope = { is_ai = no }
		}
		ai_chance = {
			base = 0 # AI will never do this
		}
		scope:subject_scope = {
			play_as = scope:overlord_scope
		}
		custom_tooltip = {
			text = frontera_ot_homeland_gained_homeland
			scope:ascendent_state = {
				set_state_owner = scope:overlord_scope
				set_state_type = unincorporated
				remove_variable = becoming_state_frontera
				state_region = { add_claim = scope:overlord_scope }
			}
		}
	}
}
