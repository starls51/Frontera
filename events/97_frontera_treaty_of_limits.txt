	cooldown = { days = -1 }

	}

	immediate = {

	}
	# Hostile Response
	option = {
		name = frontera_treaty_of_limits.2.a
		default_option = yes
		ai_chance = {
			base = 1
		}
		every_state_region = {
			limit = {
				OR = {
					region = sr:region_great_plains
					region = sr:region_pacific_coast
				}
				any_scope_state = {
					OR = {
						owner = scope:seller_scope
						owner = { is_subject_of = scope:seller_scope }
					}
				}
			}
			add_claim = scope:buyer_scope
		}
		s:STATE_FLORIDA = {
			if = {
				limit = {
					any_scope_state = {
						owner = scope:seller_scope
					}
				}
				add_claim = scope:buyer_scope
			}
		}
	}
	# Restrained Response - only for the player to try again later
	option = {
		name = frontera_treaty_of_limits.2.b
		ai_chance = {
			base = 0
		}
		remove_global_variable = frontera_treaty_of_limits_attempted
	}
}
