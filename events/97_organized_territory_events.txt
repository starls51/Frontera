namespace = frontera_ot
frontera_ot.1 = {
	type = country_event
	placement = scope:ot_scope

	title = frontera_ot.1.t
	desc = frontera_ot.1.d
	flavor = frontera_ot.1.f

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	duration = 3

	cooldown = { days = -1 }

	trigger = {
		#triggered by on_action effect when territory is too small to be viable
	}

	immediate = {
		save_scope_as = ot_scope
		overlord ?= { save_scope_as = overlord_scope }
	}

	# Become the overlord
	option = {
		name = frontera_ot.1.a
		default_option = yes
		if = {
			limit = {
				is_player = yes
			}
			play_as = scope:overlord_scope
			scope:overlord_scope = { annex = scope:ot_scope }
		}
		else = {
			scope:overlord_scope = {
				post_notification = ot_consolidated_frontera_notification
				annex = scope:ot_scope
			}
		}
	}
}
# Wagons West Event
# SCOPE = Overlord of OT
# This is an event to gain seed population and found a cultural community. The overlord gets this, and will always favor a player run OT.
frontera_ot.2 = {
	type = country_event
	placement = scope:ot_state_scope

	title = frontera_ot.2.t
	desc = frontera_ot.2.d
	flavor = frontera_ot.2.f

	event_image = {
		video = "asia_poor_people_moving"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	duration = 3

	cooldown = { days = -1 }

	trigger = {
	}

	immediate = {
		save_scope_as = overlord_scope
		random_subject_or_below = {
			limit = {
				has_law = law_type:law_frontier_territorial_administration
				NOT = { has_variable = frontera_sent_cultural_community }
			}
			save_scope_as = ot_scope
			set_variable = {
				name = frontera_sent_cultural_community
				value = yes
				years = 3
			}
			if = {
				limit = {
					any_scope_state = {
						NOT = { is_split_state = yes }
						NOT = { is_homeland_of_country_cultures = scope:overlord_scope }
					}
				}
				ordered_scope_state = {
					limit = {
						NOT = { is_split_state = yes }
						NOT = { is_homeland_of_country_cultures = scope:overlord_scope }
					}
					order_by = gdp
					position = 0
					save_scope_as = ot_state_scope
				}
			}
			else = {
				ordered_scope_state = {
					order_by = gdp
					position = 0
					save_scope_as = ot_state_scope
				}
			}
		}
		ordered_political_movement = {
			order_by = political_movement_popular_support
			position = 0
			save_scope_as = movement_a_scope
		}
		ordered_political_movement = {
			limit = {
				NOT = { scope:movement_a_scope ?= this }
			}
			order_by = political_movement_radicalism
			position = 0
			save_scope_as = movement_b_scope
		}
		if = {
			limit = {
				any_political_movement = {
					NOT = { scope:movement_a_scope ?= this }
					NOT = { scope:movement_b_scope ?= this }
				}
			}
			random_political_movement = {
				limit = {
					NOT = { scope:movement_a_scope ?= this }
					NOT = { scope:movement_b_scope ?= this }
				}
				save_scope_as = movement_c_scope
			}
		}
		else = {
			random_political_movement = { save_scope_as = movement_c_scope }
		}
		# grab relevant pops
		# if the pops arent big enough to move that many people out, just make up 500 more pops. its best this way...
		ordered_scope_pop = {
			limit = {
				pop_has_primary_culture = yes
			}
			order_by = total_size
  			position = 0
			save_scope_as = pop_a_scope
		}
		# if any pop is greater than or equal to 400, get them, else grab another
		if = {
			limit = {
				any_scope_pop = {
					total_size >= 400
					pop_has_primary_culture = yes
				}
			}
			ordered_scope_pop = {
				limit = {
					total_size >= 400
					pop_has_primary_culture = yes
				}
				order_by = pop_radical_fraction
  				position = 0
				save_scope_as = pop_b_scope
			}
		}
		else = {
			ordered_scope_pop = {
				limit = {
					pop_has_primary_culture = yes
				}
				order_by = pop_radical_fraction
  				position = 0
				save_scope_as = pop_b_scope
			}
		}
		random_scope_pop = {
			save_scope_as = pop_c_scope
		}
		if = {
			limit = {
				any_scope_pop = {
					is_pop_type = slaves
				}
			}
			ordered_scope_pop = {
				limit = {
					is_pop_type = slaves
				}
				order_by = total_size
  				position = 0
				save_scope_as = pop_slaves_scope
			}
		}
	}

	option = {
		name = frontera_ot.2.a
		scope:pop_a_scope = {
			if = {
				limit = {
					total_size >= 400
				}
				move_partial_pop = { state = scope:ot_state_scope population = 300 }
			}
			else = { move_pop = scope:ot_state_scope }
			# if pop is pro slavery, try to find 100 slaves to move with them
		}
		if = {
			limit = {
				scope:movement_a_scope = { is_political_movement_type = movement_pro_slavery }
				scope:ot_scope = { NOT = { has_law = law_type:law_slavery_banned } }
			}
			scope:pop_slaves_scope ?= {
				if = {
					limit = {
						total_size >= 150
					}
					move_partial_pop = { state = scope:ot_state_scope population = 100 }
				}
				else = { move_pop = scope:ot_state_scope }
			}
		}
		scope:movement_a_scope = {
			add_modifier = {
				name = modifier_frontier_politics_neg_frontera
				days = normal_modifier_time
			}
		}
		scope:ot_scope = {
			if = {
				limit = {
					any_political_movement = {
						scope:movement_a_scope.type ?= type
					}
				}
				random_political_movement = {
					limit = {
						scope:movement_a_scope.type ?= type
					}
					add_modifier = {
						name = modifier_frontier_politics_pos_frontera
						days = normal_modifier_time
					}
				}
			}
			else = { 
				create_political_movement = {
					type = scope:movement_a_scope.type
				}
				random_political_movement = {
					limit = {
						scope:movement_a_scope.type ?= type
					}
					add_modifier = {
						name = modifier_frontier_politics_pos_frontera
						days = normal_modifier_time
					}
				}
			}
		}
		default_option = yes
		ai_chance = {
			base = 5
		}
	}
	option = {
		name = frontera_ot.2.b
		scope:pop_b_scope = {

					}
					add_modifier = {
						name = modifier_frontier_politics_pos_frontera
						days = normal_modifier_time

				}
			}
			else = { 
				create_political_movement = {
					type = scope:movement_b_scope.type
				}
				random_political_movement = {
					limit = {
						scope:movement_b_scope.type ?= type
					}
					add_modifier = {
						name = modifier_frontier_politics_pos_frontera
						days = normal_modifier_time
					}
				}
			}
		}
		ai_chance = {
			base = 4

	}
	option = {
