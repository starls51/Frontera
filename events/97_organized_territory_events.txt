frontera_ot.1 = {
	type = country_event
	placement = scope:ot_scope

	title = frontera_ot.1.t
	desc = frontera_ot.1.d
	flavor = frontera_ot.1.f

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	duration = 3

	cooldown = { days = -1 }

	trigger = {
		#triggered by on_action effect when territory is too small to be viable
	}

	immediate = {
		save_scope_as = ot_scope
		overlord ?= { save_scope_as = overlord_scope }
	}

	# Become the overlord
	option = {
		name = frontera_ot.1.a
		default_option = yes
		if = {
			limit = {
				is_player = yes
			}
			scope:overlord_scope = { post_notification = ot_consolidated_frontera_notification }
			play_as = scope:overlord_scope
			scope:ot_scope = {
				every_scope_state = {
					set_state_owner = scope:overlord_scope
					set_state_type = unincorporated
				}
			}
		}
		else = {
			scope:overlord_scope = { post_notification = ot_consolidated_frontera_notification }
			scope:ot_scope = {
				every_scope_state = {
					set_state_owner = scope:overlord_scope
					set_state_type = unincorporated
				}
			}
		}
	}
}

frontera_ot.2 = {
	type = country_event
	placement = scope:ot_state_scope

	title = frontera_ot.2.t
	desc = frontera_ot.2.d
	flavor = frontera_ot.2.f

	event_image = {
		video = "asia_poor_people_moving"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	duration = 3

	cooldown = { days = -1 }

	trigger = {
		any_direct_subject = {
			has_law = law_type:law_frontier_territorial_administration
			has_law = law_type:law_homesteading
			any_neighbouring_state = { owner = prev.overlord }
			NOT = { has_variable = frontera_sent_cultural_community }
		}
	}

	immediate = {
		save_scope_as = overlord_scope
		random_direct_subject = {
			limit = {
				has_law = law_type:law_frontier_territorial_administration
				has_law = law_type:law_homesteading
				NOT = { has_variable = frontera_sent_cultural_community }
				any_neighbouring_state = { owner = scope:overlord_scope }
			}
			save_scope_as = ot_scope
			set_variable = {
				name = frontera_sent_cultural_community
				value = yes
				years = 3
			}
			if = {
				limit = {
					any_scope_state = {
						NOT = { is_split_state = yes }
						NOT = { is_homeland_of_country_cultures = scope:overlord_scope }
						any_neighbouring_state = { owner = scope:overlord_scope }
						NOT = {
							any_scope_pop = {
								pop_has_primary_culture = yes
							}
						}
					}
				}
				ordered_scope_state = {
					limit = {
						NOT = { is_split_state = yes }
						NOT = { is_homeland_of_country_cultures = scope:overlord_scope }
						NOT = {
							any_scope_pop = {
								pop_has_primary_culture = yes
							}
						}
					}
					order_by = arable_land
					position = 0
					save_scope_as = ot_state_scope
				}
			}
			else_if = {
				limit = {
					any_scope_state = {
						NOT = { is_split_state = yes }
						NOT = { is_homeland_of_country_cultures = scope:overlord_scope }
						any_neighbouring_state = { owner = scope:overlord_scope }
					}
				}
				ordered_scope_state = {
					limit = {
						NOT = { is_split_state = yes }
						NOT = { is_homeland_of_country_cultures = scope:overlord_scope }
					}
					order_by = arable_land
					position = 0
					save_scope_as = ot_state_scope
				}
			}

			else = {
				ordered_scope_state = {
					order_by = arable_land
					position = 0
					save_scope_as = ot_state_scope
				}
			}
		}
		ordered_political_movement = {
			order_by = political_movement_popular_support
			position = 0
			save_scope_as = movement_a_scope
		}
		ordered_political_movement = {
			limit = {
				NOT = { scope:movement_a_scope ?= this }
			}
			order_by = political_movement_radicalism
			position = 0
			save_scope_as = movement_b_scope
		}
		if = {
			limit = {
				any_political_movement = {
					NOT = { scope:movement_a_scope ?= this }
					NOT = { scope:movement_b_scope ?= this }
				}
			}
			random_political_movement = {
				limit = {
					NOT = { scope:movement_a_scope ?= this }
					NOT = { scope:movement_b_scope ?= this }
				}
				save_scope_as = movement_c_scope
			}
		}
		else = {
			random_political_movement = { save_scope_as = movement_c_scope }
		}
		# grab relevant pops
		# if the pops arent big enough to move that many people out, just make up 500 more pops. its best this way...
		ordered_scope_pop = {
			limit = {
				pop_has_primary_culture = yes
			}
			order_by = total_size
  			position = 0
			save_scope_as = pop_a_scope
		}
		# if any pop is greater than or equal to 400, get them, else grab another
		if = {
			limit = {
				any_scope_pop = {
					total_size >= 400
					pop_has_primary_culture = yes
				}
			}
			ordered_scope_pop = {
				limit = {
					total_size >= 400
					pop_has_primary_culture = yes
				}
				order_by = pop_radical_fraction
  				position = 0
				save_scope_as = pop_b_scope
			}
		}
		else = {
			ordered_scope_pop = {
				limit = {
					pop_has_primary_culture = yes
				}
				order_by = pop_radical_fraction
  				position = 0
				save_scope_as = pop_b_scope
			}
		}
		random_scope_pop = {
			save_scope_as = pop_c_scope
		}
		if = {
			limit = {
				any_scope_pop = {
					is_pop_type = slaves
				}
			}
			ordered_scope_pop = {
				limit = {
					is_pop_type = slaves
				}
				order_by = total_size
  				position = 0
				save_scope_as = pop_slaves_scope
			}
		}
	}

	option = {
		name = frontera_ot.2.a
		scope:ot_state_scope = {
			hidden_effect = {
				if = {
					limit = {
						NOT = { any_scope_culture = { scope:pop_a_scope.culture = this } }

					ordered_scope_pop = {
						order_by = total_size
  						position = 0
						change_pop_culture = {
							target = scope:pop_a_scope.culture
							value = {
								add = 1

						}
					}
				}
			}

				if = {

			move_partial_pop = {
				state = scope:ot_state_scope

					add = scope:ot_state_scope.state_population
					multiply = 0.05
					if = {
						limit = {
							scope:pop_a_scope.total_size >= 500
						}
						max = 500
					}
					else = {
						max = scope:pop_a_scope.total_size
					}
				}
			}
		}
		if = {
			limit = {
				scope:movement_a_scope = { is_political_movement_type = movement_pro_slavery }
				scope:ot_scope = { NOT = { has_law = law_type:law_slavery_banned } }
			}

				move_partial_pop = {
					state = scope:ot_state_scope

						add = scope:ot_state_scope.state_population
						multiply = 0.05
						if = {

							}
							max = 1000
						}
						else = {
							max = scope:pop_slaves_scope.total_size
						}
					}
				}
			}
		}

			add_modifier = {
				name = modifier_frontier_politics_neg_frontera
				days = normal_modifier_time
				is_decaying = yes
			}
		}
		hidden_effect = {
			scope:ot_scope = {
				if = {
					limit = {
						any_political_movement = {
							scope:movement_a_scope.type ?= type
						}
					}
					random_political_movement = {
						limit = {
							scope:movement_a_scope.type ?= type
						}
						add_modifier = {
							name = modifier_frontier_politics_pos_frontera
							days = normal_modifier_time
							is_decaying = yes
						}
					}
				}
				else = { 
					create_political_movement = {
						type = scope:movement_a_scope.type
					}
					random_political_movement = {
						limit = {
							scope:movement_a_scope.type ?= type
						}
						add_modifier = {
							name = modifier_frontier_politics_pos_frontera
							days = normal_modifier_time
						}
					}
				}
			}
		}
		default_option = yes
		ai_chance = {
			base = 5
		}

frontera_ot.3 = {
	type = country_event
	placement = scope:newly_formed_colonial_nation_anchor_state

	duration = 3

	cooldown = { days = -1 }

	trigger = {
		# triggered by create organized territory button
	}

	immediate = {
		save_scope_as = overlord_scope
		if = {
			limit = {
				any_scope_state = {
					state_region = {
						NOT = {
							any_scope_country = {
								NOT = { owner = ROOT }
								NOT = { owner = { is_subject_of = ROOT } }
							}
						}
					}
					has_decree = decree_greener_grass_campaign
					is_incorporated = no
					NOT = {is_homeland_of_country_cultures = root}
					any_neighbouring_state = {
						owner = root
						is_incorporated = yes
					}
				}
			}
			ordered_scope_state = {
				limit = {
					state_region = {
						NOT = {
							any_scope_country = {
								NOT = { owner = ROOT }
								NOT = { owner = { is_subject_of = ROOT } }
							}
						}
					}
					has_decree = decree_greener_grass_campaign

					NOT = {is_homeland_of_country_cultures = root}
					any_neighbouring_state = {
						owner = root
						is_incorporated = yes
					}
				}
				order_by = gdp
				position = 0
				check_range_bounds = no
				save_scope_as = newly_formed_colonial_nation_anchor_state
			}
		}
		else_if = {
			limit = {
				any_scope_state = {
					state_region = {
						NOT = {
							any_scope_country = {
								NOT = { owner = ROOT }
								NOT = { owner = { is_subject_of = ROOT } }
							}
						}
					}
					is_incorporated = no
					NOT = {is_homeland_of_country_cultures = root}
					any_neighbouring_state = {
						owner = root
						is_incorporated = yes
					}
				}
			}
			ordered_scope_state = {
				limit = {
					state_region = {
						NOT = {
							any_scope_country = {
								NOT = { owner = ROOT }
								NOT = { owner = { is_subject_of = ROOT } }
							}
						}
					}
					is_incorporated = no
					NOT = {is_homeland_of_country_cultures = root}
					any_neighbouring_state = {
						owner = root
						is_incorporated = yes
					}
				}
				order_by = gdp
				position = 0
				check_range_bounds = no
				save_scope_as = newly_formed_colonial_nation_anchor_state
			}
		}
		else = {
			ordered_scope_state = {
				limit = {
					state_region = {
						NOT = {
							any_scope_country = {
								NOT = { owner = ROOT }
								NOT = { owner = { is_subject_of = ROOT } }
							}
						}
					}
					is_incorporated = no
					NOT = {is_homeland_of_country_cultures = root}
					any_neighbouring_state = {
						owner = root
						is_incorporated = yes
					}
				}
				order_by = gdp
				position = 0
				check_range_bounds = no
				save_scope_as = newly_formed_colonial_nation_anchor_state
			}
		}
		# Beginning of geographic region checks for historically viable OTs
		# the order matters because some regions overlap, and OTs in regions higher up will be generated with higher priority
		scope:newly_formed_colonial_nation_anchor_state ?= {
			if = {
				limit = {
					state_region = {
						is_in_geographic_region = geographic_region_kansas_territory
					}
				}
				every_state = {
					limit = {
						is_incorporated = no
						OR = {
							owner = ROOT
							owner = {
								is_direct_subject_of = ROOT
								NOT = { can_decrease_autonomy = yes }
