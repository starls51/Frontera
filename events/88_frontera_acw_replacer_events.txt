			name = USA_wilmot_proviso

				is_insurrectionary = no

					has_ideology = ideology:ideology_anti_slavery

						is_insurrectionary = no

							leader = { has_ideology = ideology:ideology_slaver }
						}
					}
				}
				add = 8
			}
		}
		every_interest_group = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_pro_slavery
					has_ideology = ideology:ideology_pro_slavery_colonial
					leader = { has_ideology = ideology:ideology_slaver }
				}
				NOT = { has_modifier = disappointing_compromise }
			}

				name = disappointing_compromise

			limit = {
				OR = {
					has_ideology = ideology:ideology_anti_slavery
					leader = { has_ideology = ideology:ideology_abolitionist }
				}
				NOT = { has_modifier = disappointing_compromise }
			}
			add_modifier = {
				name = disappointing_compromise
				days = short_modifier_time
			}
		}

						is_insurrectionary = no

# The Emancipation Proclamation
frontera_acw_events.4 = {
	type = country_event
	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	title = acw_je_events.4.t
	desc = acw_je_events.4.d
	flavor = acw_je_events.4.f

	cooldown = { days = normal_modifier_time }

	duration = 3

	trigger = {
		# triggered on je_acw_war pulse
		exists = c:CSA
		exists = c:USA
		NOT = { has_law = law_type:law_slavery_banned }
		any_diplomatic_play = {
			AND = {
				any_scope_play_involved = { c:CSA ?= THIS }
				any_scope_play_involved = { c:USA ?= THIS }
			}
		}
		NOT = { exists = c:FSA } # fringe sanity check
		NOT = {
			any_interest_group = {
				has_modifier = delayed_emancipation
			}
		}
		has_variable = union_side_acw
	}

	immediate = {
		law_type:law_slavery_banned = {
			save_scope_as = abolition_law # use in loc with SCOPE.sLawType etc
		}
		# if a territory has slave trade, they will join the confederacy. Player can tagswitch
		every_subject_or_below = {
			limit = {
				has_law = law_type:law_frontier_territorial_administration
				has_law = law_type:law_slave_trade
			}
			if = {
				limit = {
					is_player = yes
				}
				play_as = c:CSA
			}
			save_scope_as = scope_to_remove
			c:CSA = { annex = scope:scope_to_remove }
		}
		c:CSA = {
			every_scope_state = {
				limit = {
					is_split_state = yes
				}
				state_region = {
					add_claim = c:CSA
				}
			}
		}
	}

	option = {
		name = acw_je_events.4.a # We shall save the Union!
		default_option = yes
		activate_law = law_type:law_slavery_banned
		every_subject_or_below = {
			limit = {
				has_law = law_type:law_frontier_territorial_administration
				NOT = { has_law = law_type:law_slavery_banned }
			}
			activate_law = law_type:law_slavery_banned
		}
		every_interest_group = {
			limit = {
				law_stance = {
					law = law_type:law_slavery_banned
					value > neutral
				}
			}
			add_modifier = {
				name = enacted_emancipation # +1 opinion
				days = normal_modifier_time
			}
		}
	}

	option = {
		name = acw_je_events.4.b # Pass on immediate emancipation

		every_interest_group = {
			limit = {
				law_stance = {
					law = law_type:law_slavery_banned
					value > neutral
				}
			}
			add_modifier = {
				name = delayed_emancipation # -3 opinion [everybody hated that]
				days = normal_modifier_time
			}
		}
	}
}

# Bleeding [State], a.k.a. Dynamic Bleeding Kansas
frontera_acw_events.5 = {
	type = country_event

	title = acw_events.5.t
	desc = acw_events.5.d
	flavor = acw_events.5.f
	placement = scope:bleeding_kansas

	event_image = {
		video = "europenorthamerica_american_civil_war"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	duration = 3

	trigger = {
		#exists = c:USA
		#c:USA = ROOT
		NOT = { has_law_or_variant = law_type:law_slavery_banned }
		NOT = { has_variable = bleeding_kansas_happened }
		any_subject_or_below = {
			has_law = law_type:law_frontier_territorial_administration
			NOT = { has_law = law_type:law_slavery_banned }
			any_scope_state = {
				any_scope_culture = {
					is_primary_culture_of = root.overlord
				}
				any_neighbouring_state = {
					owner = ROOT
					is_slave_state = yes
				}
			}
		}
		#any_scope_state = {
		#	is_slave_state = no
		#	is_incorporated = no
		#	any_neighbouring_state = {
		#		owner = ROOT
		#		is_slave_state = yes
		#	}
		#}
	}

	immediate = {
		set_variable = {
			name = bleeding_kansas_happened
			value = yes
		}
		random_subject_or_below = {
			limit = {
				has_law = law_type:law_frontier_territorial_administration
				NOT = { has_law = law_type:law_slavery_banned }
				any_scope_state = {
					any_scope_culture = {
						is_primary_culture_of = root.overlord
					}
					any_neighbouring_state = {
						owner = ROOT
						is_slave_state = yes
					}
				}
			}
			save_scope_as = frontera_bleeding_kansas_territory
			random_scope_state = {
			limit = {
				any_neighbouring_state = {
					owner = ROOT
					is_slave_state = yes
				}
			}
			save_scope_as = bleeding_kansas
		}
		}
	}

	cancellation_trigger = {
		is_diplomatic_play_committed_participant = yes
	}

	option = {
		name = acw_events.5.a
		hidden_effect = {
			scope:frontera_bleeding_kansas_territory = {
				activate_law = law_type:law_slavery_banned
			}
		}
		custom_tooltip = acw_events_5_free_state_tt
	}
	option = {
		name = acw_events.5.b
		default_option = yes
		scope:bleeding_kansas = {
			ordered_neighbouring_state = {
				limit = {
					owner = ROOT
					is_slave_state = yes
				}
				order_by = state_population
				position = 0
				check_range_bounds = no
				ordered_scope_pop = {
					limit = {
						is_pop_type = slaves
					}
					save_temporary_scope_as = relevant_pop
					order_by = total_size
					position = 0
					check_range_bounds = no
					move_partial_pop = {
						state = scope:bleeding_kansas
						population = {
							# move in at most 5% of the target state's population, max 5000, or the max size of the pop, whichever is smaller
							add = scope:bleeding_kansas.state_population
							multiply = 0.05
							if = {
								limit = {
									scope:relevant_pop.total_size >= 5000
								}
								max = 5000
							}
							else = {
								max = scope:relevant_pop.total_size
							}
						}
					}
				}
			}
		}
	}
}
