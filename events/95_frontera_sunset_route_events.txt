frontera_sunset_route_event.1 = {
	type = country_event
	placement = scope:california_scope
	#p:4BBEE7

	title = frontera_sunset_route_event.1.t
	desc = frontera_sunset_route_event.1.d
	flavor = frontera_sunset_route_event.1.f

	event_image = {
		video = "unspecific_trains"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	duration = 3

	cooldown = { days = -1 }

	trigger = {
		#triggered by journal entry
	}

	immediate = {
		save_scope_as = builder_scope
		random_scope_state = {
			limit = {
				state_region = s:STATE_CALIFORNIA
			}
			save_scope_as = california_scope
		}
	}

	option = {
		name = frontera_sunset_route_event.1.a
		default_option = yes
		add_modifier = {
				name = sunset_route_frontera_reward_int
				days = long_modifier_time
			}
		hidden_effect = {
			every_country = {
				limit = {
					NOT = {root = this}
					OR = {
						has_state_in_state_region = STATE_CALIFORNIA
						has_state_in_state_region = STATE_NEVADA
						has_state_in_state_region = STATE_ARIZONA
						has_state_in_state_region = STATE_NEW_MEXICO
						has_state_in_state_region = STATE_TEXAS
					}
				}
				trigger_event = { id = frontera_sunset_route_event.2 }
			}
		}
	}
}

frontera_sunset_route_event.3 = {
	type = country_event
	placement = scope:buyer_new_mexico_scope

	title = frontera_sunset_route_event.3.t
	desc = frontera_sunset_route_event.3.d
	flavor = frontera_sunset_route_event.3.f

	event_image = {
		video = "europenorthamerica_gold_prospectors"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	duration = 3

	cooldown = { days = -1 }

	trigger = {
		# triggered by button press
	}

	immediate = {
		save_scope_as = buyer_scope
		random_scope_state = {
			limit = {
				state_region = s:STATE_ARIZONA
			}
			save_scope_as = buyer_arizona_scope
		}
		random_scope_state = {
			limit = {
				state_region = s:STATE_NEW_MEXICO
			}
			save_scope_as = buyer_new_mexico_scope
		}
		ordered_country = {
			limit = {
				NOT = {root = this}
				OR = {
					has_state_in_state_region = STATE_ARIZONA
					has_state_in_state_region = STATE_NEW_MEXICO
				}
			}
			order_by = gdp
			position = 0
			check_range_bounds = no
			save_scope_as = angry_country_scope
		}
	}

	option = {
		name = frontera_sunset_route_event.3.a
		every_country = {
			limit = {
				NOT = {root = this}
				NOT = { is_subject_of = root }
				OR = {
					has_state_in_state_region = STATE_ARIZONA
					has_state_in_state_region = STATE_NEW_MEXICO
				}
			}
			trigger_event = { id = frontera_sunset_route_event.4 }
			custom_tooltip = {
				text = mesilla_treaty_creation_tt
			}
		}
		default_option = yes
	}
	option = {
		name = frontera_sunset_route_event.3.b
		every_country = {
			limit = {
				NOT = {root = this}
				NOT = { is_subject_of = root }
				OR = {
					has_state_in_state_region = STATE_ARIZONA
					has_state_in_state_region = STATE_NEW_MEXICO
				}
			}
			change_relations = {
				country = root
				value = 30
			}
			trigger_event = { id = frontera_sunset_route_event.5 }
		}
	}
}
