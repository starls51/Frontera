da_organize_territory_from_subject = {
	groups= { overlord }
	#show_in_lens = no
	requires_approval = no
	should_notify_third_parties = yes
	show_effect_in_tooltip = yes

	texture = "gfx/interface/icons/diplomatic_action_icons/doctrine_of_lapse.dds"
	#texture = "gfx/interface/icons/diplomatic_action_icons/da_appoint_colonial_governor.dds"
	confirmation_sound = "event:/SFX/UI/MapInteraction/SubjectInteractions/diplomatic_action_subjects_04_appoint_governor_benign"

	selectable = {
		has_law = law_type:law_frontier_colonization
	}

	potential = {
		has_law = law_type:law_frontier_colonization
		scope:target_country = {
			NOT = {has_law = law_type:law_frontier_territorial_administration}
			NOT = { is_diplomatic_play_enemy_of = root }
		}
	}

	possible = {
		is_revolutionary = no
		NOT = { is_country_type = unrecognized }
		NOT = {has_law = law_type:law_frontier_territorial_administration}
		scope:target_country = {
			liberty_desire <= 25
			is_revolutionary = no
			any_scope_state = {
				is_incorporated = yes
				OR = {
					custom_tooltip = {
						text = je_tooltip_homeland_required_frontera
						is_homeland_of_country_cultures = root
					}
					custom_tooltip = {
						text = je_tooltip_incorporated_adjacency_frontera
						any_neighbouring_state = {
							owner = ROOT
							is_incorporated = yes
						}
					}
				}
			}
		}
	}

	accept_effect = {
		custom_tooltip = {
			text = da_organize_territory_from_subject_effect
			scope:target_country = {
				set_country_type = colonial
				#create_diplomatic_pact = {
				#	country = root
				#	type = frontera_organized_territory
				#}
				#decrease_autonomy = yes
				#HERE would be to change starting effects based on US slavery policy choices
				#effect_starting_politics_territory_free_frontera = yes
				activate_law = law_type:law_frontier_territorial_administration
				activate_law = law_type:law_homesteading # required to activate frontier guides modifier
				activate_law = law_type:law_frontier_colonization # copy the overlords colonization charter
				add_liberty_desire = 50
			}
			hidden_effect = {
				create_diplomatic_pact = {
					country = scope:target_country
					type = frontera_organized_territory
				}
				scope:target_country = {
					random_scope_state = {
						limit = {
							is_capital = yes
						}
						save_scope_as = newly_formed_colonial_nation_anchor_state
					}
					if = {
						limit = {
							NOT = {has_modifier = colonial_administration}
						}
						add_modifier = {
							name = colonial_administration
						}
					}
					every_scope_state = {
						every_scope_culture = {
							limit = {
								has_homeland = prev
							}
							scope:target_country = {
								add_radicals = {
									value = medium_radicals
									culture = prev
								}
							}
						}
					}
					if = {
						limit = {
							root = {
								has_law = law_type:law_legacy_slavery
							}
						}
						if = {
							limit = {
								root = { has_variable = frontera_wilmot_proviso_pass }
							}
							activate_law = law_type:law_slavery_banned
						}
						else_if = {
							limit = {
								root = { has_variable = frontera_wilmot_proviso_fail }
							}
							activate_law = law_type:law_slave_trade
						}
					}
				}
				post_notification = ot_created_frontera_notification
			}
		}
	}

	ai = {
		evaluation_chance = {
			value = 0.5
		}

		propose_score = {
			value = 10
		}

		will_propose = {
			OR = {
				has_strategy = ai_strategy_colonial_expansion
				has_strategy = ai_strategy_territorial_expansion
				has_strategy = ai_strategy_nationalist_agenda
			}
		}
	}
}
