country_OT_homelands_check_frontera = {
	every_scope_state = {
		limit = {
			state_population > 85000
			any_scope_culture = {
				is_primary_culture_of = root.overlord
			}
		}
		set_variable = {
			name = accepted_percent
			value = 0.0
		}
		save_scope_as = state_to_check_scope
		every_scope_culture = {
			if = {
				limit = {
					is_primary_culture_of = root.overlord
				}
				save_scope_as = culture_to_check
				scope:state_to_check_scope = {
					# put an else statement in this to check for more cultures
					if = {
						limit = {
							culture_percent_state = {
								target = scope:culture_to_check
								value >= 0.6
							}
						}
						change_variable = {
							name = accepted_percent
							add = 0.6
						}
					}
					else_if = {
						limit = {
							culture_percent_state = {
								target = scope:culture_to_check
								value >= 0.5
							}
						}
						change_variable = {
							name = accepted_percent
							add = 0.5
						}
					}
					else_if = {
						limit = {
							culture_percent_state = {
								target = scope:culture_to_check
								value >= 0.4
							}
						}
						change_variable = {
							name = accepted_percent
							add = 0.4
						}
					}
					else_if = {
						limit = {
							culture_percent_state = {
								target = scope:culture_to_check
								value >= 0.3
							}
						}
						change_variable = {
							name = accepted_percent
							add = 0.3
						}
					}
					else_if = {
						limit = {
							culture_percent_state = {
								target = scope:culture_to_check
								value >= 0.2
							}
						}
						change_variable = {
							name = accepted_percent
							add = 0.2
						}
					}
					else_if = {
						limit = {
							culture_percent_state = {
								target = scope:culture_to_check
								value >= 0.1
							}
						}
						change_variable = {
							name = accepted_percent
							add = 0.1
						}
					}
				}
			}
		}
		if = {
			limit = {
				var:accepted_percent >= 0.6
			}
			save_scope_as = state_to_be_homeland_scope
		}
		remove_variable = accepted_percent
	}
	scope:state_to_be_homeland_scope ?= {
		ordered_scope_pop = {
			limit = {
				# I AM WORKING ON THIS PART STILL - these parts below i can try moving them, from being triggers, to effects, so i dont use temp scopes with them
				culture = { is_primary_culture_of = root.overlord }
			}
			order_by = total_size
			position = 0
			check_range_bounds = no
			culture = {
				save_temporary_scope_as = new_homelander_culture
			}
		}
		state_region = {
			add_homeland = scope:new_homelander_culture
		}
	}
}

country_OT_transfer_homeland_to_overlord = {
	if = {
		limit = {
			root.overlord = {
				is_diplomatic_play_committed_participant = no
				is_at_war = no
			}
			is_diplomatic_play_committed_participant = no
			is_at_war = no
			any_scope_state = {
				is_homeland_of_country_cultures = root.overlord
				save_temporary_scope_as = giveaway_state
			}
		}
		scope:giveaway_state = {
			trigger_event = frontera_ot_homeland.1
		}
	}
}
