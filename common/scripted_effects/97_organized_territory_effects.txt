# Maintains a list of state_regions owned by Organized Territories
state_OT_modifier_check_frontera = {
	if = {
		limit = {
			exists = THIS
			is_incorporated = yes
			owner = {
				has_government_type = gov_frontier_territory_frontera
			}
		}
		if = {
			limit = {
				has_modifier = modifier_frontierlands_frontera
			}
		}
		else = {
			add_modifier = {
				name = modifier_frontierlands_frontera
			}
		}
	}
	else = {
		remove_modifier = modifier_frontierlands_frontera
	}
}

# make any ready states within organized territories into a homeland
country_OT_homelands_check_frontera = {
	if = {
		limit = {
			any_scope_state = {
				NOT = {is_homeland_of_country_cultures = root.overlord}
				state_population > 85000
				any_scope_culture = {
					is_primary_culture_of = root.overlord
					save_temporary_scope_as = new_homelander_culture
				}
				culture_percent_state = {
		   			target = scope:new_homelander_culture
		    		value > 0.6
				}
				save_temporary_scope_as = new_homeland_state
			}
		}
		scope:new_homeland_state = {
			state_region = {
				add_homeland = scope:new_homelander_culture
			}
		}
	}
}

# make homelands in organized territories be annexed by the overlord
country_OT_transfer_homeland_to_overlord = {
	if = {
		limit = {
			root.overlord = {
				is_diplomatic_play_committed_participant = no
				is_at_war = no
			}
			is_diplomatic_play_committed_participant = no
			is_at_war = no
			any_scope_state = {
				is_homeland_of_country_cultures = root.overlord
				save_temporary_scope_as = giveaway_state
			}
		}
		scope:giveaway_state = {
			set_state_owner = root.overlord
		}
	}
}
