country_OT_homelands_check_frontera = {
	every_scope_state = {
		limit = {
			state_population > 85000
			any_scope_culture = {
				is_primary_culture_of = root.overlord
			}
			# Disables states gaining homeland if state is not fully colonized by Recognized Countries - this should be MIGRATION
			#state_region = {
			#	NOT = {
			#		any_scope_state = {
			#			owner = { is_country_type = decentralized }
			#		}
			#	}
			#}
		}
		set_variable = {
			name = accepted_percent
			value = 0.0
		}
		save_scope_as = state_to_check_scope
		every_scope_culture = {
			save_scope_as = culture_to_check
			if = {
				limit = {
					OR = {
						is_primary_culture_of = root.overlord
						AND = {
							NOT = { is_primary_culture_of = root }
							ROOT = {
								country_average_cultural_acceptance = {
									target = scope:culture_to_check
									value >= 65
								}
							}
						}
					}
				}
				scope:state_to_check_scope = {
					if = {
						limit = {
							culture_percent_state = {
								target = scope:culture_to_check
								value >= 0.6
							}
						}
						change_variable = {
							name = accepted_percent
							add = 0.6
						}
					}
					else_if = {
						limit = {
							culture_percent_state = {
								target = scope:culture_to_check
								value >= 0.5
							}
						}
						change_variable = {
							name = accepted_percent
							add = 0.5
						}
					}
					else_if = {
						limit = {
							culture_percent_state = {
								target = scope:culture_to_check
								value >= 0.4
							}
						}
						change_variable = {
							name = accepted_percent
							add = 0.4
						}
					}
					else_if = {
						limit = {
							culture_percent_state = {
								target = scope:culture_to_check
								value >= 0.3
							}
						}
						change_variable = {
							name = accepted_percent
							add = 0.3
						}
					}
					else_if = {
						limit = {
							culture_percent_state = {
								target = scope:culture_to_check
								value >= 0.2
							}
						}
						change_variable = {
							name = accepted_percent
							add = 0.2
						}
					}
					else_if = {
						limit = {
							culture_percent_state = {
								target = scope:culture_to_check
								value >= 0.1
							}
						}
						change_variable = {
							name = accepted_percent
							add = 0.1
						}
					}
				}
			}
		}
		# Inc for high pop states, to prevent migration overflow
		if = {
			limit = {
				state_population < 200000
				var:accepted_percent >= 0.6
			}
			save_scope_as = state_to_be_homeland_scope
		}
		else_if = {
			limit = {
				state_population < 500000
				var:accepted_percent >= 0.45
			}
			save_scope_as = state_to_be_homeland_scope
		}
		else_if = {
			limit = {
				state_population < 1000000
				var:accepted_percent >= 0.3
			}
			save_scope_as = state_to_be_homeland_scope
		}
		else_if = {
			limit = {
				state_population >= 1000000
				var:accepted_percent >= 0.2
			}
			save_scope_as = state_to_be_homeland_scope
		}
		remove_variable = accepted_percent
	}
	scope:state_to_be_homeland_scope ?= {
		ordered_scope_pop = {
			limit = {
				culture = { is_primary_culture_of = root.overlord }
			}
			order_by = total_size
			position = 0
			check_range_bounds = no
			culture = {
				save_temporary_scope_as = new_homelander_culture
			}
		}
		state_region = {
			add_homeland = scope:new_homelander_culture
			add_claim = root.overlord
		}
	}
}

country_OT_transfer_homeland_to_overlord = {
	if = {
		limit = {
			root.overlord = {
				is_diplomatic_play_committed_participant = no
				is_at_war = no
			}
			is_diplomatic_play_committed_participant = no
			is_at_war = no
			any_scope_state = {
				is_homeland_of_country_cultures = root.overlord
				save_temporary_scope_as = giveaway_state
			}
		}
		scope:giveaway_state = {
			trigger_event = frontera_ot_homeland.1
		}
	}
}

country_OT_consolidate_split_states = {
	root = {
		save_scope_as = ot_scope
		overlord = { save_scope_as = overlord_scope }
	}
	if = {
		limit = {
			any_scope_state = {
				is_split_state = yes
				state_region = {
					any_scope_state = {
						NOR = {
							owner = scope:overlord_scope
							owner = { is_subject_of = scope:overlord_scope }
						}
					}
				}
			}
		}
		# Do nothing
	}
	else_if = {
		limit = {
			is_diplomatic_play_committed_participant = no
			scope:overlord_scope = {
				is_diplomatic_play_committed_participant = no
			}
		}
		trigger_event = frontera_ot.1
	}
}
