REPLACE:native_american_clothes_trigger = {
	exists = scope:character
	scope:character = {
		culture = {
			NOT = {
				OR = {
					this = cu:californio_mestizo
					this = cu:tejano
				}
			}
			has_discrimination_trait_group = heritage_group_indigenous_american
		}
	}
}
