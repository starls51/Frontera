REPLACE:manifest_destiny = {
	is_shown = {
		always = no
	}
}

frontera_manifest_destiny = {
	is_shown = {
		exists = c:USA
		c:USA ?= ROOT
		NOT = { has_variable = manifested_destiny }
	}

	possible = {
		has_technology_researched = nationalism
		has_interest_marker_in_region = region_pacific_coast
		has_state_in_state_region = STATE_TEXAS
	}

	when_taken = {
		if = {
			limit = {
				country_or_subject_owns_entire_state_region = STATE_CALIFORNIA
				country_or_subject_owns_entire_state_region = STATE_NEVADA
				country_or_subject_owns_entire_state_region = STATE_UTAH
				country_or_subject_owns_entire_state_region = STATE_ARIZONA
				country_or_subject_owns_entire_state_region = STATE_NEW_MEXICO
				country_or_subject_owns_entire_state_region = STATE_TEXAS
			}
			trigger_event = { id = frontera_manifest_destiny.1 }
		}
		else = {
			add_journal_entry = {
				type = je_manifest_destiny_mexico
			}
		}
		if = {
			limit = {
				country_or_subject_owns_entire_state_region = STATE_MONTANA
				country_or_subject_owns_entire_state_region = STATE_WYOMING
				country_or_subject_owns_entire_state_region = STATE_COLORADO
				country_or_subject_owns_entire_state_region = STATE_NORTH_DAKOTA
				country_or_subject_owns_entire_state_region = STATE_SOUTH_DAKOTA
				country_or_subject_owns_entire_state_region = STATE_NEBRASKA
				country_or_subject_owns_entire_state_region = STATE_KANSAS
			}
			trigger_event = { id = frontera_manifest_destiny.2 }
		}
		else = {
			add_journal_entry = {
				type = je_manifest_destiny_frontier_wars
			}
		}
		ig:ig_landowners ?= {
			add_modifier = {
				name = ig_manifest_destiny
				days = normal_modifier_time
			}
		}
		ig:ig_intelligentsia ?= {
			add_modifier = {
				name = ig_manifest_destiny
				days = normal_modifier_time
				multiplier = -1
			}
		}
		s:STATE_OKLAHOMA = {
			if = {
				limit = {
					any_scope_state = {
						NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		s:STATE_NEBRASKA = {
			if = {
				limit = {
					any_scope_state = {
						NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		s:STATE_KANSAS = {
			if = {
				limit = {
					any_scope_state = {
						NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		s:STATE_TEXAS = {
			if = {
				limit = {
					any_scope_state = {
						NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		set_variable = manifested_destiny
		hidden_effect = {
			sr:region_pacific_coast = {
				every_scope_country = {
					limit = {
						is_subject = yes
						NOT = { is_subject_of = ROOT }
						has_law = law_type:law_frontier_territorial_administration
					}
					ROOT = {
						set_secret_goal = {
							country = PREV
							secret_goal = dominate
						}
						set_secret_goal = {
							country = PREV.overlord
							secret_goal = none
						}
					}
				}
			}
			sr:region_great_plains = {
				every_scope_country = {
					limit = {
						is_subject = yes
						NOT = { is_subject_of = ROOT }
						has_law = law_type:law_frontier_territorial_administration
					}
					ROOT = {
						set_secret_goal = {
							country = PREV
							secret_goal = dominate
						}
						set_secret_goal = {
							country = PREV.overlord
							secret_goal = none
						}
					}
				}
			}
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				country_rank < rank_value:great_power
			}
		}
		else_if = {
			limit = {
				s:STATE_CALIFORNIA = {
					any_scope_country = {
						NOT = { is_country_type = decentralized }
						relations:root >= relations_threshold:amicable
					}
				}
			}
		}
		else_if = {
			limit = {
				infamy >= 24
			}
		}
		else_if = {
			limit = {
				exists = ruler
				ruler = {
					OR = {
						has_ideology = ideology:ideology_slaver
						has_ideology = ideology:ideology_jingoist_leader
						has_ideology = ideology:ideology_fascist
					}
				}
			}
			add = 100
		}
		else_if = {
			limit = {
				has_strategy = ai_strategy_territorial_expansion
			}
			add = 100
		}
	}
}
