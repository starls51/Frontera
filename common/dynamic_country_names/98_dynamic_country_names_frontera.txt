INJECT:DEFAULT = {
	dynamic_country_name = {
		name = generic_organized_territory
		adjective = generic_organized_territory_adj

		priority = 1

		trigger = {
			exists = scope:actor
			scope:actor ?= {
				has_government_type = gov_frontier_territory_frontera
			}
		}
	}
}
