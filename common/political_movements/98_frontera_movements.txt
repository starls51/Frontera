movement_frontera_mexican_centralist = {

	category = movement_category_ideological

	ideology = ideology_frontera_mexican_centralist
	character_ideologies = {
		ideology_traditionalist
		ideology_authoritarian
		ideology_theocrat
		ideology_protectionist
		ideology_ethno_nationalist
		ideology_jingoist_leader
	}

	creation_trigger = {
		OR = {
			country_has_primary_culture = cu:mexican
			country_has_primary_culture = cu:californio
		}
		has_modifier = age_of_caudillos_modifier

	}

	disband_trigger = {
		NOT = { has_modifier = age_of_caudillos_modifier }
	}

	creation_weight = {
		value = 100
		if = {
			limit = {
				owner ?= { # If a movement of this type exists in a neighbouring country, make it more likely to spawn here as well
					any_neighbouring_state = {
						owner = {
							prestige > ROOT.owner.prestige
							any_political_movement = {
								is_political_movement_type = movement_frontera_mexican_centralist
							}
						}
					}
				}
			}
			multiply = {
				value = 2
			}
		}
	}
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}
	character_support_trigger = {
			OR = {
				has_ideology = ideology:ideology_traditionalist
				has_ideology = ideology:ideology_authoritarian
				has_ideology = ideology:ideology_integralist
				has_ideology = ideology:ideology_oligarchic
				has_ideology = ideology:ideology_theocrat
				has_ideology = ideology:ideology_jingoist_leader
			}
		}
	character_support_weight = {
		value = 200
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_authoritarian
					has_ideology = ideology:ideology_oligarchic
					has_ideology = ideology:ideology_theocrat
					has_ideology = ideology:ideology_jingoist_leader
				}
			}
			multiply = {
				value = 5
			}
		}
		if = {
			limit = {
				has_ideology = ideology:ideology_integralist
				has_ideology = ideology:ideology_traditionalist
			}
			multiply = {
				value = 2
			}
		}
	}

	pop_support_trigger = {
		NOR = {
			is_pop_type = slaves
		}
	}

	pop_support_factors = {
		movement_support_aristocrats
		movement_support_clergymen
		movement_support_shopkeepers
		movement_support_officers
		movement_support_farmers
		movement_support_clerks
		movement_support_laborers
		movement_support_soldiers
		movement_support_peasants
		movement_support_number_of_claims
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
						OR = {
							is_being_suppressed = yes
							is_being_bolstered = yes
						}
					}
				}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}

		add = {
			value = 1
			desc = "POP_BASE_SUPPORT"
		}

		if = {
			limit = {
				strata = upper
			}
			if = {
				limit = {
					is_pop_type = aristocrats
				}
				add = {
					value = 15
					desc = "POP_ARISTOCRATS"
				}
			}
			else = {
				add = {
					value = 6
					desc = "UPPER_NO_ICON"
				}
			}
		}
		else_if = {
			limit = {
				strata = middle
			}
			if = {
				limit = {
					is_pop_type = clergymen
				}
				add = {
					value = 15
					desc = "POP_CLERGYMEN"
				}
			}
			if = {
				limit = {
					is_pop_type = shopkeepers
				}
				add = {
					value = 12
					desc = "POP_SHOPKEEPERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = farmers
				}
				add = {
					value = 20
					desc = "POP_FARMERS"
				}
			}
			else = {
				add = {
					value = 9
					desc = "MIDDLE_NO_ICON"
				}
			}
		}
		else_if = {
			limit = {
				strata = lower
			}
			if = {
				limit = {
					is_pop_type = clerks
				}
				add = {
					value = 10
					desc = "POP_CLERKS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = laborers
				}
				add = {
					value = -3
					desc = "POP_LABORERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = soldiers
				}
				add = {
					value = 10
					desc = "POP_SOLDIERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = peasants
				}
				add = {
					value = 12
					desc = "POP_PEASANTS"
				}
			}
			else = {
				add = {
					value = 3
					desc = "LOWER_NO_ICON"
				}
			}
		}

		multiply = { # Illiterate pops participate at 1/10 the rate of fully literate pops
			desc = "POP_LITERACY"
			value = literacy_rate
			add = 0.2
		}

		multiply = {
			value = 0.012
			multiply = pop_acceptance
			desc = "ACCEPTANCE"
		}

		if = { # If a more radical movement exists, it'll poach radicals from this one.
			limit = {
				owner ?= {
					any_political_movement = {
						is_political_movement_type = movement_fascist
					}
				}
			}
			multiply = {
				desc = "POP_RADICAL_MOVEMENT_LOSSES"
				value = 1
				subtract = pop_radical_fraction
			}
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = {
			value = 1
			if = {
				limit = {
					OR = {
						is_capital = yes
						any_neighbouring_state = {
							is_capital = yes
						}
					}
				}
				multiply = 8
			}
			else = {
				multiply = 0.25
			}
		}
	}

	secession = {
		possible = {
			political_movement_support >= 0.05
			scope:clout <= 0
		}

		weight = {
			value = 1000
		}
	}

	additional_radicalism_factors = {
		owner ?= {
			add = {
				value = country_turmoil
				multiply = 0.5
				desc = "COUNTRY_TURMOIL"
			}
		}

		if = {
			limit = {
				owner = {
					any_neighbouring_state = {
						owner = {
							NOT = { THIS = ROOT.owner }
							any_political_movement = {
								is_political_movement_type = movement_frontera_mexican_centralist
								is_insurrectionary = yes
							}
						}
					}
				}
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
				value = 0.15
			}
		}
	}
}
