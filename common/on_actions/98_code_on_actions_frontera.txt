#Ft Ross
country_fort_ross_event_check = {
	effect = {
		# 1839 03 01 for trigger in april
		if = {
			limit = {
				game_date < 1865.4.9
				game_date > 1840.04.11
				s:STATE_CALIFORNIA = {
					NOT = { has_variable = frontera_fort_ross_sale_initiated }
					any_scope_state = {
						is_split_state = yes
					}
					NOT = {
						any_scope_state = {
							owner = {
								is_diplomatic_play_committed_participant = yes
							}
						}
					}
				}
			}
			s:STATE_CALIFORNIA = {
				set_variable = {
					name = frontera_fort_ross_sale_initiated
				}
				ordered_scope_state = {
        	        order_by = arable_land 
					position = 1
					check_range_bounds = no
        	        owner = {
						trigger_event = { id = frontera_fort_ross.1 days = -1 popup = no }
					}
				}
			}
		}
	}
}
on_yearly_pulse = {
	on_actions = {
		country_fort_ross_event_check
	}
}

#OTs
add_yearly_OT_migration_modifier = {
	effect = {
		if = {
			limit = {
				has_law = law_type:law_frontier_territorial_administration
				has_law = law_type:law_homesteading
			}
			ordered_scope_state = {
        	    order_by = gdp
				min = 0
				max = 2
				check_range_bounds = no
        	    add_modifier = {
					name = modifier_frontierlands_frontera
					months = 11
				}
			}
		}
	}
}

consolidate_split_states = {
	effect = {
		if = {
			limit = {
				has_law = law_type:law_frontier_territorial_administration
			}
		}
		country_OT_consolidate_split_states = yes
	}
}

include_OT_in_colonial_freedom = {
	effect = {
		if = {
			limit = {
				has_law = law_type:law_frontier_territorial_administration
			}
			trigger_event = { id = independence.2 days = 3 popup = yes }
			add_modifier = {
				name = modifier_foreign_administration
				days = long_modifier_time
			}
		}
	}
}

check_for_OT_homelands_frontera = {
	trigger = {
		is_subject = yes
		has_law = law_type:law_frontier_territorial_administration
	}
	effect = {
		country_OT_homelands_check_frontera = yes
	}
}

check_for_OT_giveaway_states_frontera = {
	trigger = {
		is_subject = yes
		has_law = law_type:law_frontier_territorial_administration
	}
	effect = {
		country_OT_transfer_homeland_to_overlord = yes
	}
}

# Root = Diplomatic Action
# Effect for on_action: if OT gains autonomy, they cease to be an OT and become a dominion
diplomatic_OT_liberation_frontera = {
	effect = {
		every_country = {
			limit = {
				has_law = law_type:law_frontier_territorial_administration
				is_subject = yes
				NOT = { is_subject_type = subject_type_colony }
			}
			if = {
				limit = {
					overlord = {
						has_law = law_type:law_monarchy
					}
				}
				activate_law = law_type:law_colonial_administration
			}
			else = {
				activate_law = law_type:law_presidential_republic
			}
		}
	}
}

# if OT gains autonomy, they cease to be an OT and become a dominion
on_diplomatic_action_overlord_increase_autonomy = {
	on_actions = {
		diplomatic_OT_liberation_frontera
	}
}

# Root = Diplomatic Action
on_diplomatic_action_subject_increase_own_autonomy = {
	on_actions = {
		diplomatic_OT_liberation_frontera
	}
}
