# Root = state
on_state_incorporation = {
	on_actions = {
	  check_organized_territory_list_frontera
	}
}

on_state_owner_change = {
	on_actions = {
	  check_organized_territory_list_frontera
	}
}

on_state_created = {
	on_actions = {
	  check_organized_territory_list_frontera
	}
}

check_organized_territory_list_frontera = {
	effect = {
		state_organized_territory_listcheck_frontera = yes
	}
}
check_organized_territory_list_frontera_monthly = {
	effect = {
		every_in_global_list = {
			variable = ot_state_frontera_list
		}
		state_organized_territory_listcheck_frontera = yes
	}
}
