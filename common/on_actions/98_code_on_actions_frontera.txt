#Ft Ross
country_fort_ross_event_check = {
	effect = {
		# 1839 03 01 for trigger in april
		if = {
			limit = {
				game_date < 1865.4.9
				game_date > 1840.04.11
				s:STATE_CALIFORNIA = {
					NOT = { has_variable = frontera_fort_ross_sale_initiated }
					any_scope_state = {
						is_split_state = yes
					}
					NOT = {
						any_scope_state = {
							owner = {
								is_diplomatic_play_committed_participant = yes
							}
						}
					}
				}
			}
			s:STATE_CALIFORNIA = {
				set_variable = {
					name = frontera_fort_ross_sale_initiated
				}
				ordered_scope_state = {
        	        order_by = arable_land 
					position = 1
					check_range_bounds = no
        	        owner = {
						trigger_event = { id = frontera_fort_ross.1 days = -1 popup = no }
					}
				}
			}
		}
	}
}
on_yearly_pulse = {
	on_actions = {
		country_fort_ross_event_check
	}
}

	effect = {
		if = {
			limit = {
				has_law = law_type:law_frontier_territorial_administration
			}
			trigger_event = { id = independence.2 days = 3 popup = yes }
			add_modifier = {
				name = colony_alone_modifier
				days = long_modifier_time
			}
		}
	}
}
