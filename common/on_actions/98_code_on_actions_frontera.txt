# Root = state
on_state_incorporation = {
	on_actions = {
		check_for_OT_state_modifier_frontera
	}
}

on_state_owner_change = {
	on_actions = {
		check_for_OT_state_modifier_frontera
	}
}

on_state_created = {
	on_actions = {
		check_for_OT_state_modifier_frontera
	}
}

on_game_started = {
	on_actions = {
		init_check_for_OT_state_modifier_frontera
	}
}

init_check_for_OT_state_modifier_frontera = {
	effect = {
		every_state = {
			state_OT_modifier_check_frontera = yes
		}
	}
}


check_for_OT_state_modifier_frontera = {
	effect = {
		state_OT_modifier_check_frontera = yes
	}
}

# Root = Country

on_monthly_pulse_country = {
	trigger = {
		is_subject = yes
		has_government_type = gov_frontier_territory_frontera
	}
	on_actions = {
    	check_for_OT_homelands_frontera
		check_for_OT_giveaway_states_frontera
    }
}

check_for_OT_homelands_frontera = {
	effect = {
		country_OT_homelands_check_frontera = yes
	}
}

check_for_OT_giveaway_states_frontera = {
	effect = {
		country_OT_transfer_homeland_to_overlord = yes
	}
}
