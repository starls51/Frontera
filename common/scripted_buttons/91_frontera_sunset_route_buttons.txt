je_gadsden_purchase_frontera_button = {
	name = "je_gadsden_purchase_frontera_button"
	desc = "je_gadsden_purchase_frontera_button_desc"

	visible = {
		NOT = { has_variable = gadsden_purchase_attempted }
		OR = {
			s:STATE_ARIZONA = {
				any_scope_state = {
					is_split_state = yes
					NOT = { owner ?= root }
					owner = { NOT = { is_subject_of = root } }
				}
			}
			s:STATE_NEW_MEXICO = {
				any_scope_state = {
					is_split_state = yes
					NOT = { owner ?= root }
					owner = { NOT = { is_subject_of = root } }
				}
			}
		}
	}

	cooldown = { days = 365 }

	possible = {
		OR = {
			has_state_in_state_region = STATE_ARIZONA
			has_state_in_state_region = STATE_NEW_MEXICO
		}
		NOT = {
			s:STATE_ARIZONA = {
				any_scope_state = {
					owner = {
						OR = {
							is_country_type = decentralized
							is_diplomatic_play_committed_participant = yes
						}
					}
				}
			}
		}
		NOT = {
			s:STATE_NEW_MEXICO = {
				any_scope_state = {
					owner = {
						OR = {
							is_country_type = decentralized
							is_diplomatic_play_committed_participant = yes
						}
					}
				}
			}
		}
	}

	# AI should do this
	ai_chance = {
		value = 100
	}

	effect = {
		set_variable = gadsden_purchase_attempted
		trigger_event = { id = frontera_sunset_route_event.3 popup = yes }
	}
}
