je_organize_territory_button_pacific = {
	name = "je_organize_territory_button_pacific"
	desc = "je_organize_territory_button_pacific_desc"

	visible = {}

	cooldown = { days = 35 }

	possible = {
		any_scope_state = {
			is_incorporated = no
			custom_tooltip = {
				text = je_tooltip_no_homelands_frontera
				NOT = {is_homeland_of_country_cultures = root}
			}
		}

		#prevents splitting off during diplo play or war as it invalidates wargoals. is_at_war is there for clarity for players even though it's technically included in the former
		is_diplomatic_play_committed_participant = no
		is_at_war = no

		custom_tooltip = {
			text = colonial_administration_cooldown
			NOT = {
				any_subject_or_below = {
					has_variable = newly_formed_colonial_nation_var
				}
			}
		}
	}

	ai_chance = {
		value = 100
	}

	effect = {
		custom_tooltip = {
			text = ""
			ordered_state = {
				limit = {
					owner = ROOT
					is_incorporated = no
					NOT = {is_homeland_of_country_cultures = root}
				}
				order_by = gdp
				position = 0
				check_range_bounds = no
				save_scope_as = newly_formed_colonial_nation_anchor_state
				#Beginning of if-checks for any region I would like to make Organized Territories from that are NOT strategic regions
				if = {
					limit = {
						state_region = {
							is_in_geographic_region = geographic_region_louisiana_purchase
						}
					}
					every_state = {
						limit = {
							is_incorporated = no
							NOT = {is_homeland_of_country_cultures = root}
							OR = {
								owner = ROOT
								owner = {
									is_subject_of = ROOT
									has_government_type = gov_frontier_territory_frontera
								}
							}
							is_in_geographic_region = geographic_region_louisiana_purchase
						}
						set_variable = state_to_cede
					}
				}
				# I am in the state scope of my anchor state
				else = {
					every_state = {
						limit = {
							is_incorporated = no
							NOT = {is_homeland_of_country_cultures = root}
							region = scope:newly_formed_colonial_nation_anchor_state.region
							OR = {
								owner = ROOT
								owner = {
									is_subject_of = ROOT
									has_government_type = gov_frontier_territory_frontera
								}
							}
						}
						set_variable = state_to_cede
					}
				}
			}


			if = {
				limit = {
					any_scope_state = {
						has_variable = state_to_cede
						has_decree = decree_greener_grass_campaign
					}
				}
				random_scope_state = {
					limit = {
						has_variable = state_to_cede
						has_decree = decree_greener_grass_campaign
					}
					save_scope_as = newly_formed_colonial_nation_anchor_state
				}
			}
		}
		custom_tooltip = {
			text = creating_OT_frontera_tt
			create_dynamic_country = {
				origin = root
				country_type = colonial
				tier = principality
				capital = scope:newly_formed_colonial_nation_anchor_state
				cede_state_trigger = {
					has_variable = state_to_cede
				}
				on_created = {
					effect_starting_politics_territory_free_frontera = yes
					every_scope_state = {
						add_modifier = {
							name = modifier_frontierlands_frontera
						}
					}
					set_variable = {
						name = newly_formed_colonial_nation_var
						months = 1
					}
					every_scope_state = {
						remove_variable = state_to_cede
					}
					add_modifier = {
						name = colonial_administration
					}
				}
			}
			hidden_effect = {
				create_diplomatic_pact = {
					country = scope:newly_formed_colonial_nation_anchor_state.owner
					type = colony
				}
			}
			#trigger_event = { id = colonial_administration_events.1 days = 1 popup = yes }
			#custom_tooltip = je_colonial_administration_button_tt_1
			#custom_tooltip = je_colonial_administration_button_tt_2
		}
	}
}
