REPLACE:je_oregon = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_usa_manifest_destiny

	is_shown_when_inactive = {
		exists = c:USA
		exists = c:GBR
		c:USA = ROOT
		s:STATE_WASHINGTON = {
			any_scope_state = {
				owner = {
					OR = {
						c:GBR = THIS
						is_subject_of = c:GBR
					}
				}
			}
		}
		s:STATE_OREGON = {
			any_scope_state = {
				owner = {
					OR = {
						c:GBR = THIS
						is_subject_of = c:GBR
					}
				}
			}
		}
		s:STATE_IDAHO = {
			any_scope_state = {
				owner = {
					OR = {
						c:GBR = THIS
						is_subject_of = c:GBR
					}
				}
			}
		}
	}

	possible = {
		OR = {
			has_interest_marker_in_region = region_pacific_coast
			custom_tooltip = {
				text = oregon_tt
				has_global_variable = oregon_trail_mapped
			}
		}
	}

	complete = {
		OR = {
			AND = {
				has_interest_marker_in_region = region_pacific_coast
				custom_tooltip = {
					text = oregon_tt
					has_global_variable = oregon_trail_mapped
				}
				s:STATE_IDAHO = {
					NOT = {
						any_scope_state = {
							owner = {
								is_country_type = decentralized
							}
						}
					}
				}
				s:STATE_OREGON = {
					NOT = {
						any_scope_state = {
							owner = {
								is_country_type = decentralized
							}
						}
					}
				}
				s:STATE_WASHINGTON = {
					NOT = {
						any_scope_state = {
							owner = {
								is_country_type = decentralized
							}
						}
					}
				}
				c:GBR ?= {
					relations:ROOT >= relations_threshold:cordial
				}
			}
			AND = {
				country_or_subject_owns_entire_state_region = STATE_WASHINGTON
				country_or_subject_owns_entire_state_region = STATE_IDAHO
				country_or_subject_owns_entire_state_region = STATE_OREGON
			}
		}
	}

	fail = {
		OR = {
			NOT = { has_interest_marker_in_region = region_pacific_coast }
			country_rank < rank_value:major_power
		}
		# No states in Idaho, Washington, or Oregon are decentralized
	}

	on_complete = {
		if = {
			limit = {
				s:STATE_WASHINGTON = {
					any_scope_state = {
						owner = {
							OR = {
								c:GBR = THIS
								is_subject_of = c:GBR
							}
						}
					}
				}
				s:STATE_OREGON = {
					any_scope_state = {
						owner = {
							OR = {
								c:GBR = THIS
								is_subject_of = c:GBR
							}
						}
					}
				}
				s:STATE_IDAHO = {
					any_scope_state = {
						owner = {
							OR = {
								c:GBR = THIS
								is_subject_of = c:GBR
							}
						}
					}
				}
			}
			c:GBR ?= { trigger_event = { id = frontera_oregon.2 } }
		}
		else = {
			#s:STATE_IDAHO = {
			#	add_homeland = cu:yankee
			#}
			#s:STATE_WASHINGTON = {
			#	add_homeland = cu:yankee
			#}
			#s:STATE_OREGON = {
			#	add_homeland = cu:yankee
			#}
			add_modifier = {
				name = oregon_resolved
				days = normal_modifier_time
			}
		}
	}

	weight = 100
}
