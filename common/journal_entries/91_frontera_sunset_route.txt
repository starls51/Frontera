je_sunset_route_frontera  = {
	icon = "gfx/interface/icons/invention_icons/railways.dds"

	group = je_group_usa_manifest_destiny

	is_shown_when_inactive = {
		NOT = { has_global_variable = sunset_route_built }
		has_technology_researched = railways
		has_state_in_state_region = STATE_CALIFORNIA
	}

	possible = {
		custom_tooltip = {
			text = sunset_route_built_no_tt
			NOT = { has_global_variable = sunset_route_built }
		}
		any_scope_state = {
			state_region = s:STATE_CALIFORNIA
			has_building = building_railway
			any_neighbouring_state = {
				state_region = s:STATE_ARIZONA
			}
		}
	}

	immediate = {

	}

	complete = { 
		custom_tooltip = {
			text = sunset_route_built_yes_tt
			has_global_variable = sunset_route_built
		}
	}

	on_complete = {
		hidden_effect = {
			every_state = {
				limit = {
					has_modifier = modifier_santafe_trail_frontera
				}
				remove_modifier = modifier_santafe_trail_frontera
			}
		}
		every_state = {
			limit = {
				state_region = s:STATE_CALIFORNIA
				state_region = s:STATE_NEVADA
				state_region = s:STATE_ARIZONA
				state_region = s:STATE_NEW_MEXICO
				state_region = s:STATE_TEXAS
			}
			add_modifier = {
				name = sunset_route_frontera
				days = normal_modifier_time
			}
		}
	}

	scripted_button = je_sunset_route_frontera_button

	weight = 100

	should_be_pinned_by_default = yes
}
