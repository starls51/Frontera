REPLACE:je_seminole_wars = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_usa_manifest_destiny

	complete = {
		s:STATE_FLORIDA = {
			NOT = {
				any_scope_state = {
					owner ?= { is_country_type = decentralized }
				}
			}
		}
	}

	on_complete = {
		show_as_tooltip = {
			if = {
				limit = {
					c:SEQ ?= { is_subject_of = c:USA }
					exists = s:STATE_OKLAHOMA.region_state:SEQ
				}
				custom_tooltip = {
					text = je_seminole_wars_removal
					s:STATE_FLORIDA = {
						every_scope_state = {
							every_scope_pop = {
								limit = {
									culture = cu:muskogean
								}
								move_pop = s:STATE_OKLAHOMA.region_state:SEQ
							}
						}
					}
				}
			}
			else = {
				s:STATE_FLORIDA = {
					every_scope_state = {
						add_modifier = {
							name = florida_resettlement
							days = long_modifier_time
						}
					}
				}
			}
		}
		trigger_event = { id = native_resettlement.7 }
	}

	immediate = {
		c:SML ?= {
			save_scope_as = seminoles
		}
		cu:muskogean = {
			save_scope_as = seminole_peoples
		}
		s:STATE_OKLAHOMA = {
			save_scope_as = oklahoma_region
		}
		c:SEQ ?= {
			save_scope_as = indian_territory
		}
		s:STATE_FLORIDA.region_state:SML = { save_scope_as = flarda }
	}

	fail = {
		s:STATE_FLORIDA = {
			NOT = {
				any_scope_state = {
					OR = {
						owner ?= c:USA
						owner ?= { is_subject_of = c:USA }
					}
				}
			}
		}
	}

	on_fail = {
		c:SML ?= {
			set_country_type = recognized
			hidden_effect = {
				s:STATE_FLORIDA = {
					add_claim = c:SML
					every_scope_state = {
						every_scope_pop = {
							limit = {
								OR = {
									culture = cu:afro_american
									culture = cu:muskogean
								}
							}
							move_partial_pop = { state = s:STATE_FLORIDA.region_state:SML population_ratio = 0.9 }
						}
						every_scope_pop = {
							limit = {
								OR = {
									NOT = { culture = cu:afro_american }
									NOT = { culture = cu:muskogean }
								}
							}
							move_partial_pop = { state = s:STATE_FLORIDA.region_state:SML population_ratio = 0.1 }
						}
					}
				}
				trigger_event = { id = independence.2 days = -1 popup = yes }
				add_primary_culture = cu:afro_american
				activate_law = law_type:law_migration_controls
			}
		}
		c:MSC ?= {
			set_country_type = recognized
			hidden_effect = {
				s:STATE_FLORIDA = {
					add_claim = c:MSC
					every_scope_state = {
						every_scope_pop = {
							limit = {
								OR = {
									culture = cu:afro_american
									culture = cu:muskogean
								}
							}
							move_partial_pop = { state = s:STATE_FLORIDA.region_state:MSC population_ratio = 0.9 }
						}
						every_scope_pop = {
							limit = {
								OR = {
									NOT = { culture = cu:afro_american }
									NOT = { culture = cu:muskogean }
								}
							}
							move_partial_pop = { state = s:STATE_FLORIDA.region_state:MSC population_ratio = 0.1 }
						}
					}
				}
				trigger_event = { id = independence.2 days = -1 popup = yes }
				add_primary_culture = cu:afro_american
			}
		}
		every_country = {
			limit = {
				has_claim = s:STATE_FLORIDA
			}
			save_scope_as = country_scope
			s:STATE_FLORIDA = { remove_claim = scope:country_scope }
		}
	}

	on_monthly_pulse = {
	}
	on_yearly_pulse = {
	}

	weight = 100

	should_be_pinned_by_default = yes
}
